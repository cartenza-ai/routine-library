routine_def:
  id: weekly_summary
  name: "Weekly Knowledge Summary"
  version: "1.0"
  description: >
    Rolls up all knowledge_repo entries from the past 7 days into a single
    structured summary — decisions made, issues opened, observations logged,
    and milestones hit — grouped by day. Designed to be run at the start of
    a session to re-orient quickly, or at the end of a week to review progress.

  variables:
    - name: package
      type: string
      required: true
      description: "Knowledge repo package to summarize (e.g. knowledge_partner_issues)"
    - name: days_back
      type: integer
      required: false
      default: 7
      description: "How many days back from today to include"

  pre_flight:
    - check: "package exists in knowledge_repo"
      on_fail: "abort — ask engineer to confirm package name via list_artifact_packages"

  resources:
    - type: knowledge_repo
      note: "Already connected via clone_knowledge_repo — no separate resource fetch needed"

  inputs:
    - name_pattern: "{{ package }}"
      type: knowledge_repo_package
      required: true
      bind_as: entries

  outputs:
    - name: "Weekly Summary — {{ package }} — {{ today }}"
      type: text
      package: summaries
      required: true
      bind_as: summary_artifact

  prompt_template: |
    You are the SE Knowledge Partner producing a weekly knowledge summary.

    ## Input
    Package: {{ package }}
    Window: last {{ days_back }} days, ending today

    ## Step 1 — Fetch and filter
    Use browse_knowledge_repo (or render_log_book) on {{ package }}.
    Keep only entries with a timestamp within the last {{ days_back }} days.

    ## Step 2 — Group by day
    Sort newest-first, then group entries by calendar date.

    ## Step 3 — Summarize
    For each day, write a short terse bullet list covering:
    - Decisions made (type: decision)
    - Issues opened (type: issue)
    - Observations logged (type: observation)
    - Milestones hit (type: milestone)
    Use plain, direct language. No filler, no repetition of the full entry text —
    one line per entry capturing the essential fact.

    ## Step 4 — Close with a net-status line
    End with 1-2 sentences: where things stand today and what's open.

    Keep the whole summary scannable in under a minute.

  post_execution:
    - report:
        summary_fields:
          - routine_id: "weekly_summary"
          - input_source: "{{ package }} via knowledge-repo"
          - objects_processed: "count of entries in window"
          - workspace_branch: "from create_workspace"
          - artifact_ids: "from write_workspace_artifact"
          - push_commit_sha: "from push_artifacts"
          - viewer_links: "from generate_viewer_link"
          - data_quality_flags: "omit if none"
