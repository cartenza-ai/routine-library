routine_def:
  id: consolidate_stakeholder_qa
  name: "Consolidate Stakeholder Questions & Answers"
  version: "1.0"
  description: >
    Consolidates a set of raw, grouped stakeholder-question notes (workspace text artifacts,
    one per group/session) against a source answer document, de-duplicating overlapping or
    rephrased questions — including ones buried inside narrative paragraphs — categorizing them
    by topic, and matching each to the best available answer. Produces a single merged table
    with columns #, Category, Question, Group Source, Answer, Status (Answered / Partial /
    Not addressed), so gaps needing a follow-up answer are immediately visible.

  variables:
    - name: workspace_id
      type: string
      required: true
      description: "The workspace-manager workspace_id containing the raw group question notes (e.g. Consolidate_Stakeholders_Requirements/2026-07-16)"
    - name: package
      type: string
      required: true
      description: "workspace-manager package containing the group question artifacts (e.g. 'SU II 886')"
    - name: answer_document_path
      type: string
      required: true
      description: "Path to the source answer document (e.g. an uploaded .docx of stakeholder responses)"
    - name: output_table_name
      type: string
      required: false
      default: "Consolidated Questions - With Answers"
      description: "Name for the final output table artifact"

  pre_flight:
    - check: "workspace_id exists in workspace-manager (get_workspace_status)"
      on_fail: "abort — ask engineer to confirm workspace_id"
    - check: "answer_document_path is readable"
      on_fail: "abort — ask engineer to re-upload or confirm the path"

  resources:
    - id: knowledge_repo_session
      type: knowledge_repo
      mcp_tool: clone_knowledge_repo
      note: "Already connected via clone_knowledge_repo — no separate resource fetch needed"
    - id: answer_document_file
      type: none
      mcp_tool: "n/a — local file read"
      note: "Answer document is a local file read directly (e.g. via pandoc for .docx) — not a Capella or external_api resource"

  inputs:
    - name_pattern: "Group * Questions"
      type: workspace_artifact
      required: true
      bind_as: group_notes
      note: "All text-type artifacts in {{ package }} within {{ workspace_id }} matching this name pattern, fetched via get_workspace_status + read_workspace_artifact per entry. If a name has multiple artifact_ids (no in-place update in workspace-manager), use the most recently written_at one unless the engineer has flagged a specific authoritative version."

  outputs:
    - name: "{{ output_table_name }}"
      type: table
      package: "{{ package }}"
      required: true
      bind_as: consolidated_answers_table

  prompt_template: |
    You are consolidating stakeholder requirement questions with answers from a source document.

    ## Step 1 — Gather raw questions
    Read every artifact in workspace {{ workspace_id }}, package {{ package }}, matching
    "Group * Questions". Extract every distinct question, including ones buried inside
    narrative paragraphs — read for the substantive ask, not just bullet-formatted lines.

    ## Step 2 — De-duplicate and categorize
    Merge overlapping or rephrased questions into a single canonical question. Track every
    group that raised each merged question (Group Source column, e.g. "G1, G3"). Organize the
    consolidated list into logical topic categories (e.g. Boundary & Scope, Power & Energy,
    Communications, Drilling & Core Handling, Habitability, Medical, Actors & Staffing,
    Logistics & Transport, Deliverables & Evaluation — adapt categories to the actual content).

    ## Step 3 — Match against the source answer document
    Read {{ answer_document_path }}. For each consolidated question, find the best matching
    answer:
      - Direct answer found -> summarize faithfully and concisely in your own words, mark
        Status "Answered" (note in the Answer text if only partially covered).
      - Only related/tangential context found -> summarize what's available and what's still
        missing, mark Status "Partial — fill in".
      - No relevant content found -> mark Status "Not addressed — fill in".
    An honest "not addressed" is more useful than a stretched match — do not force-fit an
    answer that isn't really there. Also flag (in the Answer text) any question that reads as
    an assignment/instructor-level question rather than something stakeholders would plausibly
    answer, rather than silently marking it not addressed.

    ## Step 4 — Output
    Produce a single table artifact named "{{ output_table_name }}" with columns:
    #, Category, Question, Group Source, Answer, Status.

  post_execution:
    - report:
        summary_fields:
          - routine_id: "consolidate_stakeholder_qa"
          - input_source: "{{ package }} / {{ workspace_id }} (group notes) + {{ answer_document_path }} (answer doc)"
          - objects_processed: "count of consolidated questions in the output table"
          - workspace_branch: "from create_workspace"
          - artifact_ids: "from write_workspace_artifact"
          - push_commit_sha: "from push_artifacts"
          - viewer_links: "from generate_viewer_link (knowledge-repo only — workspace-manager artifacts have no viewer link)"
          - data_quality_flags: "note any duplicate-named group-note artifacts encountered (workspace-manager has no in-place update — see obs-0080), and the count of questions marked Not addressed"
